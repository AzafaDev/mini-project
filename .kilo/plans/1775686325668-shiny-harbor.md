### Solusi
Edit `event.service.ts` - `getEventById` function:

```typescript
getEventById: async ({ id }: { id: string }) => {
  const event = await prisma.event.findUnique({
    where: { id, isDeleted: false },
    include: {
      organizer: {
        select: {
          id: true,
          fullName: true,
          profilePicture: true,
        },
      },
      tickets: true,
      reviews: {
        include: {
          user: {
            select: { id, fullName, profilePicture }
          }
        }
      },
      vouchers: {
        where: { 
          isActive: true, 
          startDate: { lte: new Date() }, 
          endDate: { gte: new Date() } 
        }
      },
    },
  });
  if (!event) throw new AppError("Event not found", 404);
  
  // Calculate average rating
  const avgRating = event.reviews.length > 0
    ? event.reviews.reduce((sum, r) => sum + r.rating, 0) / event.reviews.length
    : 0;
    
  return { ...event, averageRating: avgRating };
}
```
